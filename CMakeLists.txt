cmake_minimum_required(VERSION 3.20)
project(snowrunner_minimap VERSION 0.1.0 LANGUAGES CXX)
add_library(minimap_core src/minimap.cpp src/telemetry.cpp)
target_include_directories(minimap_core PUBLIC include)
target_compile_features(minimap_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(minimap_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(minimap_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(minimap_demo src/demo.cpp)
target_link_libraries(minimap_demo PRIVATE minimap_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(minimap_tests tests/minimap_tests.cpp)
  target_link_libraries(minimap_tests PRIVATE minimap_core)
  add_test(NAME minimap_geometry_and_freshness COMMAND minimap_tests)
  add_executable(telemetry_tests tests/telemetry_tests.cpp)
  target_link_libraries(telemetry_tests PRIVATE minimap_core)
  add_test(NAME native_adapter_contract COMMAND telemetry_tests)
endif()
