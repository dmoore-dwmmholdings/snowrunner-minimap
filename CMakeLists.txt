cmake_minimum_required(VERSION 3.20)
project(snowrunner_minimap VERSION 0.1.0 LANGUAGES CXX)
add_library(minimap_core src/minimap.cpp src/telemetry.cpp)
target_include_directories(minimap_core PUBLIC include)
target_compile_features(minimap_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(minimap_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(minimap_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(minimap_demo src/demo.cpp)
target_link_libraries(minimap_demo PRIVATE minimap_core)
option(SRMM_BUILD_DIAGNOSTICS "Build isolated Windows diagnostic DLL and host" ${WIN32})
if(SRMM_BUILD_DIAGNOSTICS)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Diagnostics require x64 Windows; disable SRMM_BUILD_DIAGNOSTICS elsewhere.")
  endif()
  add_library(srmm_diagnostic SHARED src/diagnostic_plugin.cpp)
  target_link_libraries(srmm_diagnostic PRIVATE minimap_core)
  add_executable(srmm_diagnostic_host tools/diagnostic_host.cpp)
  target_include_directories(srmm_diagnostic_host PRIVATE include)
  target_compile_features(srmm_diagnostic_host PRIVATE cxx_std_17)
  add_dependencies(srmm_diagnostic_host srmm_diagnostic)
  if(MSVC)
    target_compile_options(srmm_diagnostic PRIVATE /W4 /permissive-)
    target_compile_options(srmm_diagnostic_host PRIVATE /W4 /permissive-)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(minimap_tests tests/minimap_tests.cpp)
  target_link_libraries(minimap_tests PRIVATE minimap_core)
  add_test(NAME minimap_geometry_and_freshness COMMAND minimap_tests)
  add_executable(telemetry_tests tests/telemetry_tests.cpp)
  target_link_libraries(telemetry_tests PRIVATE minimap_core)
  add_test(NAME native_adapter_contract COMMAND telemetry_tests)
  if(SRMM_BUILD_DIAGNOSTICS)
    add_test(NAME isolated_diagnostic_plugin COMMAND srmm_diagnostic_host)
  endif()
  if(WIN32)
    find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
    add_test(NAME pe_identity_inspection COMMAND "${POWERSHELL_EXECUTABLE}"
      -NoProfile -NonInteractive -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/pe_image_tests.ps1")
  endif()
endif()
