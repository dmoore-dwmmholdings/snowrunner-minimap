cmake_minimum_required(VERSION 3.20)
project(srmm_dxgi_route LANGUAGES CXX)
file(TO_NATIVE_PATH "$ENV{SystemRoot}/System32/dxgi" SRMM_SYSTEM_DXGI)
string(REPLACE "\\" "\\\\" SRMM_SYSTEM_DXGI "${SRMM_SYSTEM_DXGI}")
configure_file(dxgi_exports.inc.in dxgi_exports.inc @ONLY)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Owned DXGI route currently requires MSVC x64 Windows")
endif()
add_library(srmm_dxgi_bridge SHARED dxgi_bridge.cpp dxgi_bridge.def shared_renderer.cpp map_dds.cpp bootstrap_probe.cpp)
target_link_libraries(srmm_dxgi_bridge PRIVATE d3d11 d3dcompiler bcrypt)
target_include_directories(srmm_dxgi_bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(srmm_dxgi_bridge PROPERTIES OUTPUT_NAME dxgi ARCHIVE_OUTPUT_NAME srmm_dxgi_exports)
add_executable(srmm_dxgi_host dxgi_host.cpp)
add_executable(srmm_alignment_host alignment_host.cpp shared_renderer.cpp map_dds.cpp candidate_packet.cpp)
target_link_libraries(srmm_alignment_host PRIVATE d3d11 d3dcompiler dxgi)
target_compile_features(srmm_alignment_host PRIVATE cxx_std_17)
target_compile_options(srmm_alignment_host PRIVATE /W4 /permissive-)
add_dependencies(srmm_alignment_host srmm_dxgi_bridge)
target_link_libraries(srmm_dxgi_host PRIVATE d3d11 dxgi)
add_dependencies(srmm_dxgi_host srmm_dxgi_bridge)
foreach(target srmm_dxgi_bridge srmm_dxgi_host)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE /W4 /permissive-)
endforeach()
include(CTest)
option(SRMM_BUILD_GAME_PROBE "Build separate fingerprint-gated game bootstrap probe; no game drawing or deployment" OFF)
if(SRMM_BUILD_GAME_PROBE)
  add_library(srmm_dxgi_game_probe SHARED dxgi_bridge.cpp dxgi_bridge.def shared_renderer.cpp map_dds.cpp bootstrap_probe.cpp)
  target_compile_features(srmm_dxgi_game_probe PRIVATE cxx_std_17)
  target_compile_options(srmm_dxgi_game_probe PRIVATE /W4 /permissive-)
  target_compile_definitions(srmm_dxgi_game_probe PRIVATE SRMM_ENABLE_GAME_PROBE=1)
  target_include_directories(srmm_dxgi_game_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(srmm_dxgi_game_probe PRIVATE d3d11 d3dcompiler bcrypt)
  set_target_properties(srmm_dxgi_game_probe PROPERTIES OUTPUT_NAME dxgi ARCHIVE_OUTPUT_NAME srmm_dxgi_game_exports RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/game-probe")
  add_dependencies(srmm_dxgi_game_probe srmm_dxgi_host)
  add_custom_command(TARGET srmm_dxgi_game_probe POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:srmm_dxgi_host>" "$<TARGET_FILE_DIR:srmm_dxgi_game_probe>/srmm_dxgi_host.exe")
  add_test(NAME owned_game_probe_build COMMAND "$<TARGET_FILE_DIR:srmm_dxgi_game_probe>/srmm_dxgi_host.exe")
  set_tests_properties(owned_game_probe_build PROPERTIES TIMEOUT 20)
endif()
add_test(NAME owned_dxgi_com_route COMMAND srmm_dxgi_host)
add_test(NAME owned_map_orientation COMMAND srmm_alignment_host)
set_tests_properties(owned_map_orientation PROPERTIES TIMEOUT 20)
set_tests_properties(owned_dxgi_com_route PROPERTIES TIMEOUT 20)
